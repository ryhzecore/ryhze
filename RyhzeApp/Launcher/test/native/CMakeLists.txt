cmake_minimum_required(VERSION 3.20)
project(RyhzeWindowsChecks LANGUAGES CXX)
enable_testing()
add_executable(steam_session_test steam_session_test.cpp ../../windows/runner/steam_session.cpp)
target_compile_features(steam_session_test PRIVATE cxx_std_17)
target_compile_definitions(steam_session_test PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(steam_session_test PRIVATE /W4 /WX /EHsc)
target_link_libraries(steam_session_test PRIVATE user32 advapi32)
add_test(NAME steam_window_ownership COMMAND steam_session_test)
set_tests_properties(steam_window_ownership PROPERTIES TIMEOUT 20)
